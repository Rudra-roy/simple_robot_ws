  Parameter              Default    Range          What it does
  ─────────────────────────────────────────────────────────────────────
  min_obstacle_z         0.05m      0.0 - 1.0m    🔧 MOST IMPORTANT!
                                                   Ground filter height
                                                   ↑ if ground = obstacles
                                                   
  inflation_radius       0.3m       0.0 - 2.0m    Safety buffer
                                                   ↓ for tighter nav
                                                   ↑ for safety
                                                   
  publish_frequency      15 Hz      1 - 50 Hz     Update rate
                                                   ↓ to reduce CPU
                                                   
  obstacle_max_range     10.0m      1 - 30m       Sensor range limit
  
  cost_scaling_factor    10.0       1 - 50        Cost gradient
                                                   ↓ = gentler gradient


🔧 COMMON PROBLEMS & SOLUTIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  ❌ Ground shows as red obstacles
     ✅ Increase min_obstacle_z → try 0.1, 0.2, 0.3...
     
  ❌ Real obstacles not detected  
     ✅ Decrease min_obstacle_z or increase max_obstacle_z
     
  ❌ Robot won't go through gaps
     ✅ Decrease inflation_radius → try 0.2, 0.15
     
  ❌ Costmap lagging/slow
     ✅ Decrease publish_frequency → try 10Hz or 5Hz
     
  ❌ Too many false obstacles
     ✅ Increase min_obstacle_z
     ✅ Decrease obstacle_max_range


💻 COMMAND LINE QUICK COMMANDS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  # Check current value
  ros2 param get /costmap_node min_obstacle_z
  
  # Set new value
  ros2 param set /costmap_node min_obstacle_z 0.15
  
  # List all parameters
  ros2 param list /costmap_node
  
  # Save current settings
  ros2 param dump /costmap_node > my_params.yaml


📊 DEBUGGING WORKFLOW
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  1. Launch with GUI:
     ros2 launch nav2_costmap_node costmap_with_gui.launch.py
     
  2. Open RViz and add:
     - /costmap (OccupancyGrid)  
     - /zed/zed_node/point_cloud/cloud_registered (PointCloud2)
     
  3. Adjust min_obstacle_z in GUI while watching RViz
     
  4. Fine-tune inflation_radius and other params
     
  5. Save optimal settings:
     ros2 param dump /costmap_node > optimal_params.yaml


🎯 OPTIMAL TUNING TIPS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
